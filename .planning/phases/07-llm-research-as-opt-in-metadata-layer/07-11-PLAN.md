    import hashlib
    import json
    import zipfile

    def test_zip_diff_limited_to_zip_bound_fields_when_toggling_strict_tolerant(
        iberia_project_with_full_overlay, tmp_path, monkeypatch
    ):
        from medieval_forge.services.research import overlay as overlay_module
        from medieval_forge.services.export.zip import build_unity_zip

        # Build zip A under Strict
        monkeypatch.setattr(overlay_module, "_ZIP_BOUND_FIELDS", frozenset({"name"}))
        zip_a = tmp_path / "export_strict.zip"
        build_unity_zip(iberia_project_with_full_overlay, zip_a)

        # Build zip B under Tolerant
        monkeypatch.setattr(
            overlay_module, "_ZIP_BOUND_FIELDS",
            frozenset({"name", "kingdom_owner", "historical_notes"})
        )
        zip_b = tmp_path / "export_tolerant.zip"
        build_unity_zip(iberia_project_with_full_overlay, zip_b)

        # (a) Every file EXCEPT territory_metadata.json + manifest.json byte-identical
        with zipfile.ZipFile(zip_a) as za, zipfile.ZipFile(zip_b) as zb:
            files_a = sorted(za.namelist())
            files_b = sorted(zb.namelist())
            assert files_a == files_b, f"file list differs: A={files_a} B={files_b}"
            for fname in files_a:
                if fname in ("territory_metadata.json", "manifest.json"):
                    continue
                sha_a = hashlib.sha256(za.read(fname)).hexdigest()
                sha_b = hashlib.sha256(zb.read(fname)).hexdigest()
                assert sha_a == sha_b, f"REVIEWS fix #9: geometric output {fname} differs between Strict and Tolerant"

            # (b) territory_metadata.json diff limited to _ZIP_BOUND_FIELDS
            meta_a = json.loads(za.read("territory_metadata.json"))
            meta_b = json.loads(zb.read("territory_metadata.json"))
            assert len(meta_a["condados"]) == len(meta_b["condados"])
            for ca, cb in zip(meta_a["condados"], meta_b["condados"]):
                # Same id + same geometric fields
                assert ca["id"] == cb["id"]
                for invariant_field in ("original_idx", "pixel_count", "pixel_center", "lon", "lat", "duchy", "kingdom"):
                    assert ca[invariant_field] == cb[invariant_field], \
                        f"REVIEWS fix #9: {invariant_field} differs for condado {ca['id']} between Strict and Tolerant"
                # name should match in BOTH (Strict emits name; Tolerant emits name)
                assert ca["name"] == cb["name"]
                # kingdom_owner + historical_notes should be NONE/MISSING in A, populated in B (if overlay had them)
                # No assertion that they are populated — only that they don't leak into A
                if ca.get("kingdom_owner") is not None:
                    pytest.fail(f"REVIEWS fix #9: Strict zip leaked kingdom_owner for {ca['id']}")
                if ca.get("historical_notes") is not None:
                    pytest.fail(f"REVIEWS fix #9: Strict zip leaked historical_notes for {ca['id']}")

            # (c) Both manifests have research_overlay_applied = true
            man_a = json.loads(za.read("manifest.json"))
            man_b = json.loads(zb.read("manifest.json"))
            assert man_a["research_overlay_applied"] is True
            assert man_b["research_overlay_applied"] is True
