       ```python
       """SQLite-backed research cache (D-11).

       Key derivation per RESEARCH §Example 2 + §Pitfall 6: canonical form is
       `{country_qid_lower}|{period_label_strip_lower}|{provider}|{model}|{prompt_version}`.
       SHA-256 hex digest gives a fixed-width 64-char key.

       Re-ingestion does NOT invalidate the cache (D-24 carried from v1).
       Force-refresh checkbox in UI dialog bypasses cache.
       """
       from __future__ import annotations

       import hashlib
       import json

       from sqlalchemy import select
       from sqlalchemy.ext.asyncio import AsyncSession

       from ..models import ResearchCache as ResearchCacheModel

       # Bump this when prompt.py changes meaningfully — prevents stale cache hits.
       # The literal-port prompt.py (Plan 03) is "v1"; if a future plan rewrites parts,
       # bump to "v2".
       PROMPT_VERSION = "v1"


       def cache_key(
           country_qid: str,
           period_label: str,
           provider: str,
           model: str,
           prompt_version: str = PROMPT_VERSION,
       ) -> str:
           """Derive a deterministic SHA-256 cache key.

           Normalizes country_qid + period_label to lowercase + stripped (Pitfall 6).
           Provider + model + prompt_version case is preserved (model strings are
           often case-sensitive on the upstream API).
           """
           canonical = (
               f"{country_qid.strip().lower()}|"
               f"{period_label.strip().lower()}|"
               f"{provider}|"
               f"{model}|"
               f"{prompt_version}"
           )
           return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


       async def cache_get(session: AsyncSession, key: str) -> dict | None:
           row = await session.scalar(
               select(ResearchCacheModel).where(ResearchCacheModel.cache_key == key)
           )
           if row is None:
               return None
           return json.loads(row.payload) if isinstance(row.payload, str) else row.payload


       async def cache_put(
           session: AsyncSession,
           key: str,
           payload: dict,
           provider: str,
           model: str,
       ) -> None:
           existing = await session.scalar(
               select(ResearchCacheModel).where(ResearchCacheModel.cache_key == key)
           )
           if existing:
               existing.payload = payload
               existing.provider = provider
               existing.model = model
           else:
               session.add(ResearchCacheModel(
                   cache_key=key, payload=payload, provider=provider, model=model
               ))
           await session.commit()
       ```

    1. Create `backend/medieval_forge/services/research/runner.py`. Mirror `api/v3/generate.py` structure:

       ```python
       """Research orchestration — SSE producer/consumer.

       Mirrors api/v3/generate.py exactly:
       - asyncio.Queue[str | None] with None sentinel termination
       - per-(project_id) _RUN_QUEUES single-flight gate (409 if alive)
       - finally: pop _RUN_QUEUES + _RUN_TASKS (WR-02 carry from Phase 04.1)

       Research-specific:
       - 4 conceptual stages: kingdoms -> duchies -> condados -> baronies
       - Cache check before provider call (D-11)
       - On success: write project_dir/research_overlay.json atomically (Pitfall 1)
       """
       from __future__ import annotations

       import asyncio
       import json
       from pathlib import Path

       from sqlalchemy.ext.asyncio import AsyncSession

       from ..llm import PROVIDERS, build_map_research_prompt, MapResearchResult, run_with_retry
       from ..paths import project_dir
       from .cache import cache_key, cache_get, cache_put, PROMPT_VERSION
       from .matcher import build_pipeline_condado_list, llm_output_to_overlay
       from .overlay import merge_overlay  # not used here; overlay merge is in Plan 08 zip writer

       _RUN_QUEUES: dict[str, asyncio.Queue[str | None]] = {}
       _RUN_TASKS: dict[str, asyncio.Task] = {}


       def _emit(queue, event_type: str, stage: str, message: str = "", progress: float | None = None):
           """Envelope matches api/v3/generate.py:_emit — Discretion #5 + RESEARCH §Example 1."""
           payload = {
               "event_type": event_type,
               "stage": stage,
               "message": message,
               "progress": progress,
           }
           queue.put_nowait(f"data: {json.dumps(payload)}\n\n")


       def _write_overlay_atomic(overlay_path: Path, overlay: dict) -> None:
           """Pitfall 1 + Example 3 — atomic write via tmp+replace."""
           tmp = overlay_path.with_suffix(overlay_path.suffix + ".tmp")
           tmp.write_text(json.dumps(overlay, indent=2), encoding="utf-8")
           tmp.replace(overlay_path)


       async def start_research(
           project_id: str,
           country_qid: str,
           period_label: str,
           provider_id: str,
           model: str,
           force_refresh: bool,
           db_session: AsyncSession,
           db_payload: dict | None,
       ) -> str:
           """Start a research run. Returns run_id (== project_id for single-flight).

           Raises HTTPException(409) if a run is already in flight for this project.
           """
           if project_id in _RUN_QUEUES:
               from fastapi import HTTPException
               raise HTTPException(status_code=409, detail="research run already in flight for this project")
           queue: asyncio.Queue[str | None] = asyncio.Queue()
           _RUN_QUEUES[project_id] = queue
           task = asyncio.create_task(_producer(
               project_id, country_qid, period_label, provider_id, model,
               force_refresh, db_session, db_payload, queue,
           ))
           _RUN_TASKS[project_id] = task
           return project_id


       async def _producer(
           project_id, country_qid, period_label, provider_id, model,
           force_refresh, db_session, db_payload, queue,
       ):
           """Producer task. Writes overlay on success; always puts None sentinel; WR-02 evictions."""
           try:
               _emit(queue, "started", "init", f"Starting research: {country_qid} / {period_label}")

               # Cache check (D-11)
               key = cache_key(country_qid, period_label, provider_id, model)
               if not force_refresh:
                   cached = await cache_get(db_session, key)
                   if cached is not None:
                       _emit(queue, "cached", "init", "Result from cache", progress=1.0)
                       # Write the overlay from cached payload, emit terminal success
                       overlay_path = project_dir(project_id) / "research_overlay.json"
                       _write_overlay_atomic(overlay_path, cached)
                       _emit(queue, "terminal", "done", "Pesquisa concluída (cache)", progress=1.0)
                       return

               # Load metadata for prompt injection
               metadata_path = project_dir(project_id) / "output" / "territory_metadata.json"
               metadata = json.loads(metadata_path.read_text(encoding="utf-8"))
               pipeline_condado_list = build_pipeline_condado_list(metadata)
               pipeline_condado_ids = {c["id"] for c in pipeline_condado_list}

               # Get provider and call
               provider = PROVIDERS.get(provider_id)
               if provider is None:
                   _emit(queue, "error", "init", f"Unknown provider: {provider_id}")
                   return

               _emit(queue, "stage_start", "kingdoms", "Iniciando pesquisa de reinos")
               # ... per-stage emits interleaved with provider.research() call ...
               # For V1 simplicity: one provider call, all 4 stages emit as it progresses.

               prompt = build_map_research_prompt(
                   country_qid=country_qid,
                   period_label=period_label,
                   condados=pipeline_condado_list,
               )

               llm_result_model = await run_with_retry(
                   lambda p, **_: provider.research(p, MapResearchResult, db_payload, queue),
                   prompt, MapResearchResult, max_retries=3,
               )

               _emit(queue, "stage_end", "baronies", "Resultado validado", progress=0.9)

               # Matcher: collapse to overlay shape
               overlay = llm_output_to_overlay(llm_result_model.model_dump(), pipeline_condado_ids)

               # Atomic write (Pitfall 1)
               overlay_path = project_dir(project_id) / "research_overlay.json"
               _write_overlay_atomic(overlay_path, overlay)

               # Cache the result
               await cache_put(db_session, key, overlay, provider_id, model)

               _emit(queue, "terminal", "done", f"Pesquisa concluída — {len(overlay)} condados", progress=1.0)
           except Exception as exc:
               _emit(queue, "error", "init", f"{type(exc).__name__}: {exc}")
           finally:
               # None sentinel terminates consumer
               queue.put_nowait(None)
               # WR-02 — evict so next /start works; late /stream returns 404 (intentional)
               _RUN_QUEUES.pop(project_id, None)
               _RUN_TASKS.pop(project_id, None)


       async def get_stream(project_id: str):
           """Consumer-side generator. 404 if no active run (intentional WR-02 behavior)."""
           from fastapi import HTTPException
           queue = _RUN_QUEUES.get(project_id)
           if queue is None:
               raise HTTPException(status_code=404, detail="no active research run; POST /research/start to begin")
           while True:
               item = await queue.get()
               if item is None:
                   break
               yield item


       async def stop_research(project_id: str) -> bool:
           """Best-effort cancellation. Returns True if a task was found and cancelled."""
           task = _RUN_TASKS.get(project_id)
           if task and not task.done():
               task.cancel()
               return True
           return False
       ```

       ```python
       """api/v3/research.py — research SSE orchestration endpoints.

       Endpoints:
         POST   /v3/research/start            - body {project_id, country_qid, period, provider, model, force_refresh}
         GET    /v3/research/stream/{run_id}  - SSE consumer (run_id == project_id for single-flight)
         POST   /v3/research/stop/{run_id}    - best-effort cancellation
         GET    /v3/research/providers        - list PROVIDERS + per-provider health
         GET    /v3/research/health           - alias for /providers (returns same shape)
         GET    /projects/{project_id}/research/overlay - raw overlay JSON if present, 404 otherwise

       Router prefix: /v3 (main.py adds /api at mount time per Phase 05 STATE convention).
       """
       from __future__ import annotations

       import asyncio
       import json
       from pathlib import Path

       from fastapi import APIRouter, Depends, HTTPException
       from fastapi.responses import StreamingResponse, JSONResponse
       from pydantic import BaseModel
       from sqlalchemy.ext.asyncio import AsyncSession

       from ..database import get_session
       from ..services.credential_store import get_credentials
       from ..services.llm import PROVIDERS
       from ..services.paths import project_dir
       from ..services.research.runner import start_research, get_stream, stop_research

       router = APIRouter(prefix="/v3/research", tags=["research"])


       class StartResearchBody(BaseModel):
           project_id: str
           country_qid: str
           period: str
           provider: str
           model: str
           force_refresh: bool = False


       @router.post("/start")
       async def start(body: StartResearchBody, db: AsyncSession = Depends(get_session)):
           # Get db_payload for this provider (None if not stored)
           db_payload = await get_credentials(db, body.provider)
           run_id = await start_research(
               project_id=body.project_id,
               country_qid=body.country_qid,
               period_label=body.period,
               provider_id=body.provider,
               model=body.model,
               force_refresh=body.force_refresh,
               db_session=db,
               db_payload=db_payload,
           )
           return {"run_id": run_id}


       @router.get("/stream/{run_id}")
       async def stream(run_id: str):
           return StreamingResponse(get_stream(run_id), media_type="text/event-stream")


       @router.post("/stop/{run_id}")
       async def stop(run_id: str):
           cancelled = await stop_research(run_id)
           return {"cancelled": cancelled}


       @router.get("/providers")
       async def providers(db: AsyncSession = Depends(get_session)):
           """List PROVIDERS with health-check + configured-bool.

           Never returns the credential payload (T-07-04-02 mitigation).
           """
           result = []
           for provider_id, provider in PROVIDERS.items():
               db_payload = await get_credentials(db, provider_id)
               health = await provider.health_check(db_payload)
               result.append({
                   "provider_id": provider_id,
                   "display_name": provider.display_name,
                   "healthy": health.healthy,
                   "message": health.message,
                   "configured": db_payload is not None,
                   # Never include `db_payload` itself
               })
           return result


       @router.get("/health")
       async def health(db: AsyncSession = Depends(get_session)):
           return await providers(db)


       # Separate endpoint mounted under /v3/projects/{project_id}/research/overlay
       overlay_router = APIRouter(prefix="/v3/projects", tags=["research"])


       @overlay_router.get("/{project_id}/research/overlay")
       async def get_overlay(project_id: str):
           """Returns raw overlay JSON or 404."""
           overlay_path = project_dir(project_id) / "research_overlay.json"
           if not overlay_path.exists():
               raise HTTPException(status_code=404, detail="no overlay for this project")
           return JSONResponse(content=json.loads(overlay_path.read_text(encoding="utf-8")))
       ```

       ```python
       class CredentialPayload(BaseModel):
           type: str = "api_key"

       @router.get("")
       async def list_all(db: AsyncSession = Depends(get_session)):
           """Return per-provider {configured: bool}; never the payload itself."""
           result = {}
           for provider_id in ("claude", "ollama"):
               row = await get_credentials(db, provider_id)
               result[provider_id] = {"configured": row is not None}
           return result


       @router.post("/{provider}")
       async def store(provider: str, body: CredentialPayload, db: AsyncSession = Depends(get_session)):
           await store_credentials(db, provider, body.model_dump())
           return {"stored": True}


       @router.delete("/{provider}")
       async def clear(provider: str, db: AsyncSession = Depends(get_session)):
           await clear_credentials(db, provider)
           return {"cleared": True}
       ```
