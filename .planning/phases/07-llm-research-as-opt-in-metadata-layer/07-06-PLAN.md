    2. Suggested body:

       ```python
       """Matcher — bridge LLM MapResearchResult to overlay shape.

       Two pure functions:
       - build_pipeline_condado_list: prompt-injection source (id + centroid)
       - llm_output_to_overlay: result-collapse + unknown-id drop

       D-03 field semantics (Pattern 5): kingdom_owner = parent kingdom's historical name
       (e.g., "Reino de Asturias"); distinct from condado.kingdom (geometric cluster id).
       """
       from __future__ import annotations

       from typing import Iterable


       def build_pipeline_condado_list(metadata: dict) -> list[dict]:
           """Extract {id, lon, lat} per condado for LLM prompt injection."""
           return [
               {"id": c["id"], "lon": c["lon"], "lat": c["lat"]}
               for c in metadata["condados"]
           ]


       def llm_output_to_overlay(
           llm_result: dict,
           pipeline_condado_ids: Iterable[str],
       ) -> dict:
           """Collapse MapResearchResult hierarchy into flat overlay shape.

           Drops any LLM-output condado id NOT in pipeline_condado_ids (defensive —
           retry loop should prevent unknowns via schema, but belt-and-braces).
           """
           allowed = set(pipeline_condado_ids)
           overlay: dict = {}
           for kingdom in llm_result.get("kingdoms", []):
               kingdom_owner = kingdom.get("name", "")
               for duchy in kingdom.get("duchies", []):
                   for condado in duchy.get("condados", []):
                       cid = condado.get("id")
                       if cid not in allowed:
                           continue
                       overlay[cid] = {
                           "name": condado.get("name"),
                           "kingdom_owner": kingdom_owner,
                           "historical_notes": condado.get("historical_notes"),
                       }
           return overlay
       ```
