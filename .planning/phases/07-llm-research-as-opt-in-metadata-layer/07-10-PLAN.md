       ```typescript
       import { useMutation, useQueryClient } from '@tanstack/react-query'

       export interface ValidationErrorEntry {
         code: string
         severity: 'error' | 'warning'
         file: string | null
         context: Record<string, unknown>
         message: string
       }

       export interface ValidationEnvelope {
         detail: {
           summary: string
           errors: ValidationErrorEntry[]
           warnings: ValidationErrorEntry[]
         }
       }

       export class ExportValidationError extends Error {
         envelope: ValidationEnvelope
         constructor(envelope: ValidationEnvelope) {
           super(envelope.detail?.summary ?? 'Export validation failed')
           this.name = 'ExportValidationError'
           this.envelope = envelope
         }
       }

       export interface ExportResponse {
         zip_path: string
         sha256: string
         size_bytes: number
         // ... whatever Phase 06 export.py returns; adapt
       }

       export function useExportV3(projectId: string | undefined) {
         const qc = useQueryClient()
         return useMutation({
           mutationFn: async (params: { dryRun?: boolean } = {}) => {
             const url = params.dryRun
               ? `/api/v3/projects/${projectId}/export?dry_run=true`
               : `/api/v3/projects/${projectId}/export`
             const res = await fetch(url, { method: 'POST' })
             if (res.status === 422) {
               const envelope = await res.json() as ValidationEnvelope
               throw new ExportValidationError(envelope)
             }
             if (!res.ok) throw new Error(`EXPORT_FAILED: ${res.status}`)
             // dry-run returns JSON report; real export returns the zip stream
             if (params.dryRun) return await res.json()
             // For real export: planner picks browser-download mechanism per UI-SPEC §Interaction Contract
             // Recommend URL.createObjectURL(blob) + <a download> shim for cross-browser:
             const blob = await res.blob()
             return blob
           },
           onSuccess: () => {
             qc.invalidateQueries({ queryKey: ['projects', projectId] })
             qc.invalidateQueries({ queryKey: ['projects'] })
           },
         })
       }
       ```
