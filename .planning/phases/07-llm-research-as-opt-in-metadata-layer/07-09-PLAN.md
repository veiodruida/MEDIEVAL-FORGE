       export interface ProviderEntry {
         provider_id: string
         display_name: string
         healthy: boolean
         message: string
         configured: boolean
       }

       export function useProviders(enabled = true) {
         return useQuery<ProviderEntry[]>({
           queryKey: ['v3', 'research', 'providers'],
           queryFn: async () => {
             const res = await fetch('/api/v3/research/providers')
             if (!res.ok) throw new Error(`providers fetch failed: ${res.status}`)
             return res.json()
           },
           enabled,

         })
       }

    2. Create `frontend/src/api/useResearchOverlay.ts` — UI-SPEC §Interaction Contract permits full-payload + client-derive. Pick that variant (saves a backend route):

       ```typescript
       import { useQuery } from '@tanstack/react-query'

       export interface ResearchOverlayMeta {
         exists: boolean
         provider?: string
         model?: string
         created_at?: string
         covered_condado_ids: string[]
       }

       /**
        * Derives summary metadata from the raw overlay payload (CONTEXT D-04 endpoint).
        * Strategy: fetch the full overlay; on 404 return {exists:false, covered_condado_ids:[]};
        * on 200 return {exists:true, covered_condado_ids:Object.keys(payload)}.
        *
        * Per UI-SPEC, this is the "client-derive" implementation. A future v3.1 plan
        * can add a `/research/overlay/meta` endpoint and swap this hook to consume it.
        */
       export function useResearchOverlay(projectId: string | undefined) {
         return useQuery<ResearchOverlayMeta>({
           queryKey: ['v3', 'projects', projectId, 'research', 'overlay'],
           queryFn: async () => {
             const res = await fetch(`/api/v3/projects/${projectId}/research/overlay`)
             if (res.status === 404) return { exists: false, covered_condado_ids: [] }
             if (!res.ok) throw new Error(`overlay fetch failed: ${res.status}`)
             const payload = await res.json()
             return {
               exists: true,
               covered_condado_ids: Object.keys(payload),
               // provider/model/created_at not present in raw overlay; left undefined.
               // If UI-SPEC microcopy "Última pesquisa: ..." needs these, add a meta endpoint.
             }
           },
           enabled: !!projectId,
           staleTime: 30_000,
         })
       }
       ```
