The defective function (manual_edit.py:327-387):
```python
def replay_vertex_ring(
    barony_name: str,
    barony_name_to_idx: dict[str, int],
    vertices_dict: dict[str, dict],
    polygons_by_id: dict[int, list[Polygon]],
    cfg: RegionConfig,
    W: int,
    H: int,
) -> None:
    raster_idx = barony_name_to_idx.get(barony_name)
    if raster_idx is None:
        return
    ring_keys = sorted([k for k in vertices_dict if k.startswith(barony_name + "#")],
                       key=lambda x: int(x.split("#")[1]) if "#" in x else 0)
    if len(ring_keys) < 3:
        return
    pixel_coords = [_lonlat_to_rasterio_xy(vertices_dict[k]["lon"], vertices_dict[k]["lat"], cfg, W, H)
                    for k in ring_keys]
    try:
        poly = Polygon(pixel_coords)
        if not poly.is_valid:
            poly = poly.buffer(0)        # <-- can return MultiPolygon; NOT guarded
    except Exception:
        return
    if not poly.is_empty and poly.area > 0:
        polygons_by_id[raster_idx] = [poly]   # <-- stores MultiPolygon directly (DEFECT)
```

```python
def build_baronies_geojson_sidecar(
    *,                              # keyword-only — `result=` MUST be passed by name
    result: np.ndarray,            # the rasterised int16 array (e.g. out2)
    bars: list[dict],              # index-aligned; bars[bi]["name"] is the feature id
    cfg: RegionConfig,
    barony_cmap: dict[str, int],   # feeds barony_colors.json ONLY — pass {} in the test
    out_dir: str,
) -> tuple[str, str]:              # writes baronies.geojson + barony_colors.json
    res32 = result.astype(np.int32)
    shapes_per_idx = {}
    for geom, idx in rasterio.features.shapes(res32, mask=(res32 >= 0)):   # ← the real extraction
        shapes_per_idx.setdefault(int(idx), []).append(shape(geom))
    # a Feature is emitted for bars[bi] iff shapes_per_idx.get(bi) is non-empty
    #   features.append({"type": "Feature", "id": bars[bi]["name"], ...})
```
This extraction is `res32 = result.astype(np.int32)` then
`rasterio.features.shapes(res32, mask=(res32 >= 0))` WITHOUT a transform — materially different
from an unmasked `shapes(out2)` over an int16 array. Test 1's gate calls THIS function and asserts
Gijón's Feature survives in the emitted `baronies.geojson` — the real 12-file-export artifact,
not a proxy.

The sidecar writer reads the rasterised output downstream; a barony absent from the raster is
