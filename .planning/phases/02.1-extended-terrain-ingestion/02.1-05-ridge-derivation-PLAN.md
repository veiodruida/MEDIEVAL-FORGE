    ```python
    """Phase 2.1 GEO-07: derive ridge polygons + centerlines from DEM.

    Pipeline (D-14):
      1. Read DEM (rasterio).
      2. Gaussian smooth (denoise crests).
      3. Slope (degrees) via numpy.gradient + arctan(hypot).
      4. Negative laplacian (convex-up = ridge candidate) via scipy.ndimage.laplace.
      5. Threshold mask = (slope > s_min) AND (lap > c_min) AND (z != nodata).
      6. Skeletonize via skimage.morphology.skeletonize (Pitfall 6: bool input).
      7. Polygonize the buffered skeleton → MultiPolygon (adjacency penalty zone).
      8. Extract centerline from skeleton → LineString.
      9. Build feature with deterministic id = sha1(string-format) (Pitfall 7).
     10. Cross-ref OSM peaks (raw/topography.geojson) for peak_count + naming (D-16, D-19).
     11. Cross-ref HydroSHEDS basins (raw/basins.geojson) for basin_ids[] tagging.
    """
    from __future__ import annotations
    import json
    import logging
    from hashlib import sha1
    from pathlib import Path
    from typing import Any
    import numpy as np
    import rasterio
    from rasterio.features import shapes as raster_shapes
    from scipy import ndimage
    from skimage import morphology
    from shapely.geometry import (GeometryCollection, LineString, MultiLineString,
                                  MultiPolygon, Point, Polygon, mapping, shape as shp_shape)

    log = logging.getLogger(__name__)

    # Calibrated against backend/tests/fixtures/synthetic_dem.tif (Wave 0 Task 0).
    # If you change these, re-run the calibration probe and update both numbers + this comment.
    THRESHOLDS: dict[str, dict[str, float]] = {
        "low":  {"slope_min_deg": 25.0, "curvature_min": 0.0010, "min_polygon_area_m2": 1.0e6},
        "med":  {"slope_min_deg": 30.0, "curvature_min": 0.0020, "min_polygon_area_m2": 5.0e6},
        "high": {"slope_min_deg": 35.0, "curvature_min": 0.0030, "min_polygon_area_m2": 1.0e7},
    }

    def _check_high_quality_available() -> None:
        """D-14: 'high' sensitivity requires `whitebox` extra. Raise if missing."""
        try:
            import whitebox  # noqa: F401
        except ImportError as exc:
            raise RuntimeError(
                "sensitivity='high' requires the [high-quality] extra. "
                "Install with: pip install medieval-forge[high-quality]"
            ) from exc

    def _stable_ridge_id(centroid_x: float, centroid_y: float, elev_max: float) -> str:
        """Pitfall 7: string-format hashing — deterministic across Python/numpy versions."""
        key = f"{centroid_x:.3f}|{centroid_y:.3f}|{int(round(elev_max))}"
        return sha1(key.encode("ascii")).hexdigest()[:12]

    def derive_ridges(
        dem_path: Path,
        sensitivity: str = "med",
        topography_path: Path | None = None,
        basins_path: Path | None = None,
    ) -> dict[str, Any]:
        if sensitivity not in THRESHOLDS:
            raise ValueError(f"sensitivity must be one of {list(THRESHOLDS)}; got {sensitivity!r}")
        if sensitivity == "high":
            _check_high_quality_available()
        cfg = THRESHOLDS[sensitivity]

        with rasterio.open(dem_path) as src:
            z = src.read(1).astype("float32")
            nodata = src.nodata if src.nodata is not None else -32768
            transform = src.transform

        valid = z != nodata
        z_smooth = ndimage.gaussian_filter(np.where(valid, z, 0).astype("float32"), sigma=1.5)
        dy, dx = np.gradient(z_smooth)
        slope_deg = np.degrees(np.arctan(np.hypot(dx, dy)))
        # Normalize laplacian by smoothed magnitude scale to make curvature_min scale-aware.
        lap = -ndimage.laplace(z_smooth) / max(z_smooth.std(), 1.0)
        mask = (slope_deg > cfg["slope_min_deg"]) & (lap > cfg["curvature_min"]) & valid
        # Pitfall 6: bool input.
        skeleton = morphology.skeletonize(mask.astype(bool))

        # Polygonize the mask itself (gives the buffered ridge zone).
        polys: list[Polygon] = []
        for geom_dict, value in raster_shapes(mask.astype("uint8"), mask=mask, transform=transform):
            if value != 1:
                continue
            poly = shp_shape(geom_dict)
            # Approximate area: 1 deg² ≈ 12321 km² at equator → 1.2321e10 m². Convert min area to deg².
            min_area_deg2 = cfg["min_polygon_area_m2"] / (111_000.0**2)
            if poly.area >= min_area_deg2:
                polys.append(poly)

        # Optional OSM peak cross-ref (D-16/D-19).
        peaks: list[tuple[Point, str | None]] = []
        if topography_path is not None and Path(topography_path).exists():
            try:
                topo = json.loads(Path(topography_path).read_text(encoding="utf-8"))
                for f in topo.get("features", []):
                    if f["properties"].get("natural") != "peak":
                        continue
                    geom = f.get("geometry", {})
                    if geom.get("type") == "Point":
                        x, y = geom["coordinates"]
                        peaks.append((Point(x, y), f["properties"].get("name")))
            except Exception as exc:
                log.warning("topography cross-ref failed (continuing without peaks): %s", exc)

        # Optional HydroSHEDS basin cross-ref (D-19 basin_ids[]).
        basins: list[tuple[Polygon | MultiPolygon, int]] = []
        if basins_path is not None and Path(basins_path).exists():
            try:
                bj = json.loads(Path(basins_path).read_text(encoding="utf-8"))
                for f in bj.get("features", []):
                    g = shp_shape(f["geometry"])
                    bid = int(f["properties"].get("id") or 0)
                    basins.append((g, bid))
            except Exception as exc:
                log.warning("basins cross-ref failed (continuing without basin_ids): %s", exc)

        features = []
        for poly in polys:
            cx, cy = poly.centroid.x, poly.centroid.y
            # Sample DEM under polygon (rough: bbox window read).
            mn, mx, mean = _elev_stats(poly, z, transform, nodata)
            contained_peaks = [(pt, name) for (pt, name) in peaks if poly.contains(pt)]
            peak_count = len(contained_peaks)
            # Naming: pick the most-frequent name root (or first non-null name).
            named = [n for (_, n) in contained_peaks if n]
            ridge_name = named[0] if named else None
            basin_ids = sorted({bid for (g, bid) in basins if poly.intersects(g)})

            # Centerline: extract skeleton pixels inside polygon → vectorize as LineString(s).
            centerline = _skeleton_to_line(skeleton, poly, transform)

            feat_id = _stable_ridge_id(cx, cy, mx)
            feature = {
                "type": "Feature",
                "properties": {
                    "id": feat_id,
                    "name": ridge_name,
                    "elev_min": float(mn),
                    "elev_max": float(mx),
                    "elev_mean": float(mean),
                    "peak_count": int(peak_count),
                    "basin_ids": basin_ids,
                },
                "geometry": mapping(GeometryCollection([MultiPolygon([poly]), centerline])),
            }
            features.append(feature)

        return {"type": "FeatureCollection", "features": features}

    def _elev_stats(poly: Polygon, z: np.ndarray, transform, nodata: float) -> tuple[float, float, float]:
        # Approximate: sample integer cells inside poly bbox and filter by contains.
        from rasterio.features import geometry_mask
        height, width = z.shape
        m = geometry_mask([poly], transform=transform, out_shape=(height, width), invert=True)
        sample = z[m & (z != nodata)]
        if sample.size == 0:
            return 0.0, 0.0, 0.0
        return float(sample.min()), float(sample.max()), float(sample.mean())

    def _skeleton_to_line(skeleton: np.ndarray, poly: Polygon, transform) -> LineString | MultiLineString:
        """Cheap-but-correct: take all skeleton pixels inside poly, ordered by row then col,
        and connect via LineString. For complex skeletons this is a polyline approximation —
        acceptable for v1 visualization (D-15 — centerline is for visualization; the polygon
        does the adjacency-penalty work)."""
        from rasterio.features import geometry_mask
        height, width = skeleton.shape
        m = geometry_mask([poly], transform=transform, out_shape=(height, width), invert=True)
        ys, xs = np.where(skeleton & m)
        if len(xs) < 2:
            # Degenerate: return a tiny line at centroid so geometry stays valid.
            cx, cy = poly.centroid.x, poly.centroid.y
            return LineString([(cx, cy), (cx + 1e-6, cy + 1e-6)])
        # Convert pixel (col,row) to world (x,y) via transform.
        coords = [transform * (int(c), int(r)) for r, c in zip(ys, xs)]
        return LineString(coords)
    ```

    from medieval_forge.services.ingest_terrain import ridges as _ridges

    async def run_terrain_ridges(
        project_id: str,
        queue: asyncio.Queue[str | None],
        *,
        sensitivity: str = "med",
        stop_event: asyncio.Event | None = None,
    ) -> None:
        if stop_event is None:
            stop_event = register_stop_event(project_id, "ridges")
        try:
            raw_dir = ensure_project_dirs(project_id)["raw"]
            dem_path = raw_dir / "dem.tif"
            if not dem_path.exists():
                await queue.put("data: ERROR: raw/dem.tif não existe — execute o passo DEM primeiro.\n\n")
                return
            await queue.put(f"data: derivando ridges (sensitivity={sensitivity})...\n\n")
            t0 = time.monotonic()
            try:
                fc = await asyncio.to_thread(
                    _ridges.derive_ridges, dem_path, sensitivity,
                    raw_dir / "topography.geojson", raw_dir / "basins.geojson",
                )
            except RuntimeError as exc:
                # D-14: high-quality not installed → 412.
                if "high-quality" in str(exc):
                    await queue.put(f"data: 412: {exc}\n\n")
                    return
                raise
            elapsed = time.monotonic() - t0
            if elapsed > 60.0:
                # D-18 fail-soft: warn but don't abort.
                await queue.put(f"data: AVISO: derivação levou {elapsed:.1f}s (>60s budget). Continuando.\n\n")
            _write_geojson_atomic(raw_dir / "ridges.geojson", fc)
            await queue.put(f"data: ridges: {len(fc['features'])} features → ridges.geojson\n\n")
            await queue.put("data: DONE\n\n")
        except asyncio.CancelledError:
            await queue.put("data: Cancelado pelo usuário.\n\n")
        except Exception as exc:  # noqa: BLE001
            logger.exception("run_terrain_ridges failed")
            await queue.put(f"data: ERROR: {exc.__class__.__name__}\n\n")
        finally:
            clear_stop_event(project_id, "ridges")
            await queue.put(None)
