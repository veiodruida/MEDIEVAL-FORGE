    ```typescript
    import type {
      ValidationIssue,
      GeoJSONPolygon,
      GeoJSONMultiPolygon,
      Position,
      ProjectGeometryState,
    } from '../types/editing'

    /**
     * Per-operation validation (D-06). Affects only touched territories + their
     * direct neighbors. Not a full map scan — that lives in Phase 6 VALIDATE-01..07.
     */
    export function validateTerritories(
      affectedIds: string[],
      state: ProjectGeometryState,
    ): ValidationIssue[] {
      const issues: ValidationIssue[] = []

      for (const id of affectedIds) {
        const geom = state.territories[id]
        if (!geom) continue

        // empty_polygon
        if (geom.type === 'Polygon') {
          const exterior = geom.coordinates[0] ?? []
          if (exterior.length === 0) {
            issues.push({ condado_id: id, severity: 'error', rule: 'empty_polygon',
              message: 'Território sem geometria' })
            continue
          }

          // polygon_invalid — pragmatic checks (<4 coords or duplicate consecutive)
          if (exterior.length < 4) {
            issues.push({ condado_id: id, severity: 'error', rule: 'polygon_invalid',
              message: 'Polígono inválido: menos de 4 vértices' })
            continue
          }
          let hasDuplicate = false
          for (let i = 1; i < exterior.length; i++) {
            if (exterior[i][0] === exterior[i - 1][0] && exterior[i][1] === exterior[i - 1][1]) {
              hasDuplicate = true
              break
            }
          }
          if (hasDuplicate) {
            issues.push({ condado_id: id, severity: 'error', rule: 'polygon_invalid',
              message: 'Polígono inválido: vértices duplicados consecutivos' })
            continue
          }

          // capital_outside
          const capital = state.capitals[id]
          if (capital && !pointInPolygon(capital, exterior)) {
            issues.push({ condado_id: id, severity: 'error', rule: 'capital_outside',
              message: 'Capital fora do território' })
          }
        }
      }
      return issues
    }

    /** Ray-casting point-in-polygon test for a single ring. */
    function pointInPolygon(point: Position, ring: Position[]): boolean {
      const [x, y] = point
      let inside = false
      for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) {
        const [xi, yi] = ring[i]
        const [xj, yj] = ring[j]
        const intersect = yi > y !== yj > y &&
          x < ((xj - xi) * (y - yi)) / (yj - yi + 1e-12) + xi
        if (intersect) inside = !inside
      }
      return inside
    }
    ```

    - User edits in auto → status = 'saving' immediately, then 'saved' 1.5s after last edit (debounce).

    - User edits in explicit → status = 'unsaved' until Ctrl+S flush succeeds → status = 'saved'.

    ```typescript
    import { create } from 'zustand'
    import type { SaveStrategy, SaveStatus } from '../types/editing'
    import { useProjectStore } from '../stores/useProjectStore'

    const LS_KEY = 'mf:save-strategy'
    const DEBOUNCE_MS = 1500

    function loadStrategy(): SaveStrategy {
      try {
        const v = localStorage.getItem(LS_KEY) as SaveStrategy | null
        if (v === 'auto' || v === 'per_op' || v === 'explicit') return v
      } catch {}
      return 'auto'
    }
    function saveStrategyToLS(s: SaveStrategy) {
      try { localStorage.setItem(LS_KEY, s) } catch {}
    }

    interface PersistenceState {
      strategy: SaveStrategy
      status: SaveStatus
      lastEditAt: number | null
      setStrategy: (s: SaveStrategy) => void
      markEdit: () => void
      markSaved: () => void
      markUnsaved: () => void
    }

    export const usePersistenceStore = create<PersistenceState>()((set) => ({
      strategy: loadStrategy(),
      status: 'saved',
      lastEditAt: null,
      setStrategy: (s) => {
        saveStrategyToLS(s)
        set({ strategy: s, status: s === 'explicit' ? 'unsaved' : 'saved' })
      },
      markEdit: () => set({ lastEditAt: Date.now(), status: 'saving' }),
      markSaved: () => set({ status: 'saved', lastEditAt: null }),
      markUnsaved: () => set({ status: 'unsaved' }),
    }))

    /**
     * Call from every edit finalize callback. Behavior depends on strategy:
     * - auto: set status=saving, schedule a 1.5s debounce to flip to saved
     * - per_op: immediately set status=saved (backend already persisted per call)
     * - explicit: set status=unsaved, wait for user Ctrl+S
     */
    let debounceTimer: ReturnType<typeof setTimeout> | null = null

    export function onOperationFinalized() {
      const { strategy, markEdit, markSaved, markUnsaved } = usePersistenceStore.getState()
      if (strategy === 'auto') {
        markEdit()
        if (debounceTimer) clearTimeout(debounceTimer)
        debounceTimer = setTimeout(() => markSaved(), DEBOUNCE_MS)
      } else if (strategy === 'per_op') {
        markSaved()
      } else {
        markUnsaved()
      }
    }

    /** Ctrl+S flush (explicit mode). Sends the full in-memory snapshot to
     * POST /api/geometry/save. No-op in auto/per_op (already persisted per op).
     */
    export async function manualSave(): Promise<void> {
      const { strategy, markSaved, markUnsaved } = usePersistenceStore.getState()
      if (strategy !== 'explicit') return
      const { projectId, territories, capitals } = useProjectStore.getState()
      if (!projectId) return
      const { saveSnapshot } = await import('../api/edit')
      try {
        await saveSnapshot(projectId, { territories, capitals })
        markSaved()
      } catch (err) {
        console.error('manualSave failed', err)
        markUnsaved()  // remain unsaved; user can retry
      }
    }

    export const useSaveStatus = () => usePersistenceStore((s) => s.status)
    export const useSaveStrategy = () => usePersistenceStore((s) => s.strategy)
    export const setStrategy = (s: SaveStrategy) => usePersistenceStore.getState().setStrategy(s)

    // Initialize (call once from app root)
    export function initPersistence() {
      const s = loadStrategy()
      usePersistenceStore.setState({
        strategy: s,
        status: s === 'explicit' ? 'saved' : 'saved',  // start clean
      })
    }
    ```

    import { useEffect } from 'react'
    import { useSaveStatus, useSaveStrategy } from '../services/persistence'

    export function useBeforeUnloadGuard() {
      const status = useSaveStatus()
      const strategy = useSaveStrategy()
      useEffect(() => {
        if (strategy !== 'explicit' || status !== 'unsaved') return
        const handler = (e: BeforeUnloadEvent) => {
          e.preventDefault()
          e.returnValue = '' // browser shows default confirmation
        }
        window.addEventListener('beforeunload', handler)
        return () => window.removeEventListener('beforeunload', handler)
      }, [status, strategy])
    }
