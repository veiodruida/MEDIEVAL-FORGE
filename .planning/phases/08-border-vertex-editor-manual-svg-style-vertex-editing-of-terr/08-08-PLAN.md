```tsx
import { Box, Heading, RadioGroup, Badge, Button, Flex } from '@radix-ui/themes';
import { useEditorStore } from '@/stores/useEditorStore';

export const LandmaskEditorHeader: React.FC<{
  projectId: string;
  branchId: string;
  onApply: () => Promise<void>;
}> = ({ projectId, branchId, onApply }) => {
  const mode = useEditorStore(s => s.landmaskMode);
  const setMode = useEditorStore(s => s.setLandmaskMode);
  return (
    <Box>
      <Heading size="2">Landmask</Heading>
      <RadioGroup.Root value={mode} onValueChange={(v) => setMode(v as 'manual'|'auto-immediate')}>
        <RadioGroup.Item value="manual">Manual (Aplicar para regenerar)</RadioGroup.Item>
        <RadioGroup.Item value="auto-immediate">Auto imediato (~10s/edição)</RadioGroup.Item>
      </RadioGroup.Root>
      {mode === 'auto-immediate' && (
        <Badge color="orange" variant="soft">Modo auto: ~10s por edição</Badge>
      )}
      {mode === 'manual' && (
        <Button variant="outline" color="blue" onClick={onApply}>
          Aplicar landmask
        </Button>
      )}
    </Box>
  );
};
```
