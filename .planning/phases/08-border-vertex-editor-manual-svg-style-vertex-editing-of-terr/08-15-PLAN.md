    In frontend/src/components/canvas/VertexEditLayer.tsx, add a useEffect (after
    visibleEntries is declared) that — gated on `import.meta.env.DEV` — installs a window
    function returning current editor state AND the first visible handle's page-space
    screen position (so the UAT can drive a real Konva drag):
    ```ts
    useEffect(() => {
      if (!import.meta.env.DEV) return
      ;(window as unknown as { __forgeEditorState?: () => unknown }).__forgeEditorState = () => {
        const stage = stageRef.current
        let firstHandle: { id: string; x: number; y: number } | null = null
        if (stage) {
          // find the first rendered vertex handle Circle and resolve its page coords
          const circle = stage.findOne((n: import('konva').Konva.Node) =>
            n.getClassName?.() === 'Circle' && Boolean((n.getAttr('data-vertex-id'))))
          if (circle) {
            const abs = circle.getAbsolutePosition()           // includes stage scale+pos
            const rect = stage.container().getBoundingClientRect()
            firstHandle = {
              id: String(circle.getAttr('data-vertex-id')),
              x: rect.left + abs.x,
              y: rect.top + abs.y,
            }
          }
        }
        return {
          activeTerritoryId,
          vertexCount: Object.keys(vertices).length,
          visibleHandleCount: isLandmaskMode ? 0 : visibleEntries.length,
          editLogLength: useEditorStore.getState().editLog.length,
          activeTool: useEditorStore.getState().activeTool,
          firstHandle,   // { id, x, y } page coords for page.mouse, or null
        }
      }
      return () => {
        delete (window as unknown as { __forgeEditorState?: unknown }).__forgeEditorState
      }
    }, [activeTerritoryId, vertices, isLandmaskMode, visibleEntries, stageRef])
    ```
    Mirror BaronyLayer's DEV-gated pattern exactly (cleanup on unmount). Ships nothing in
    production. If `stage.findOne` predicate typing is awkward, use
    `stage.find('Circle')` and pick the first node carrying a `data-vertex-id` attr.
  </action>
