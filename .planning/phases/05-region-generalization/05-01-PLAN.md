    ```python
    import re, yaml
    from pathlib import Path
    from pydantic import BaseModel, ConfigDict, Field
    from .contracts import RegionConfig, ProjectDataset

    _REGION_KEY_RE = re.compile(r'^[a-z0-9_]+$')
    _REGION_CACHE: dict[str, RegionConfig] = {}
    _DEFAULT_REGIONS_DIR = Path(__file__).resolve().parents[4] / "data" / "regions"

    class DatasetSchema(BaseModel):
        # R-03 (review): all dataset fields optional so template-only regions
        # (e.g. england_1216) load through pydantic and surface as a clean
        # FileNotFoundError("template-only") instead of a ValidationError.
        model_config = ConfigDict(extra='forbid')
        pt_geojson: str | None = None
        es_input: str | None = None
        mountain_river_json: str | None = None
        dem_raster: str | None = None

    class TerritorySchema(BaseModel):
        model_config = ConfigDict(extra='forbid')
        id: str
        name: str
        original_idx: int
        # ...mirror territory_data.py fields (kingdom, duchy, color, etc.)

    class RegionConfigSchema(BaseModel):
        model_config = ConfigDict(extra='forbid')
        name: str
        display_name: str | None = None  # R-06 (review) — human-readable label
        map_w: int = 1920
        map_h: int = 1080
        upscale: int = 2
        lon_min: float; lon_max: float; lat_min: float; lat_max: float
        output_dir: str = "../Assets/StreamingAssets/Maps"
        dataset: DatasetSchema = Field(default_factory=DatasetSchema)  # R-03 (review)
        border_polygon: list[list[float]] = Field(default_factory=list)
        pt_duchies: list[str] = Field(default_factory=list)
        kingdom_colors: dict[str, list[int]] = Field(default_factory=dict)
        # cleanup
        island_min_px: int = 300
        fragment_min_px: int = 600
        blob_merge_px: int = 200
        median_passes: int = 8
        smooth_sigma: float = Field(default=3.0, ge=3.0, le=4.5)  # CLAUDE.md rule 2
        # mountain/coast/etc — mirror all RegionConfig fields verbatim

        draw_names: bool = False
        kingdoms: list[dict] = Field(default_factory=list)
        duchies: list[dict] = Field(default_factory=list)
        condados: list[TerritorySchema] = Field(default_factory=list)

    def clear_region_cache() -> None:
        _REGION_CACHE.clear()

    def load_region(key: str, regions_dir: Path | None = None) -> RegionConfig:
        if not _REGION_KEY_RE.match(key):
            raise ValueError(f"invalid region key: {key!r}")
        if key in _REGION_CACHE:
            return _REGION_CACHE[key]
        rdir = regions_dir or _DEFAULT_REGIONS_DIR
        yaml_path = rdir / f"{key}.yaml"
        if not yaml_path.exists():
            raise FileNotFoundError(f"region YAML not found for key={key!r}: {yaml_path}")
        with yaml_path.open("r", encoding="utf-8") as f:
            raw = yaml.safe_load(f)
        model = RegionConfigSchema.model_validate(raw)
        # Resolve dataset paths YAML-relative under {rdir}/{key}/
        region_root = (rdir / key).resolve()
        def _resolve(p: str) -> Path:
            resolved = (region_root / p).resolve()
            try:
                resolved.relative_to(region_root)
            except ValueError:
                raise ValueError(f"dataset path {p!r} resolves outside region dir {region_root}")
            if not resolved.exists():
                raise FileNotFoundError(f"dataset file missing: {resolved}")
            return resolved
        # R-03 (review): template-only regions (e.g. england_1216) declare a
        # YAML with NO dataset fields populated, or the inputs/ directory simply
        # does not exist on disk. Surface this as a clear FileNotFoundError that
        # callers can catch and present as "region template available; dataset
        # not shipped".
        ds = model.dataset
        all_unset = (ds.pt_geojson is None and ds.es_input is None and ds.mountain_river_json is None)
        if all_unset:
            raise FileNotFoundError(f"Region {key!r} has no dataset on disk (template-only)")
        try:
            dataset = ProjectDataset(
                pt_geojson=_resolve(ds.pt_geojson) if ds.pt_geojson else None,
                es_input=_resolve(ds.es_input) if ds.es_input else None,
                mountain_river_json=_resolve(ds.mountain_river_json) if ds.mountain_river_json else None,
                dem_raster=_resolve(ds.dem_raster) if ds.dem_raster else None,
            )
        except FileNotFoundError as exc:
            # Re-raise with the template-only hint when the YAML names files that
            # do not exist (matches Plan 05-09 England behavior).
            raise FileNotFoundError(f"Region {key!r} has no dataset on disk (template-only): {exc}") from exc
        # Build kwargs for RegionConfig from model dump, swap dataset, autogen if empty
        kwargs = model.model_dump()
        kwargs['dataset'] = dataset
        kwargs['pt_duchies'] = set(kwargs.get('pt_duchies') or [])
        kwargs['kingdom_colors'] = {k: tuple(v) for k, v in (kwargs.get('kingdom_colors') or {}).items()}
        if not kwargs.get('condados'):
            kwargs['kingdoms'], kwargs['duchies'], kwargs['condados'] = _autogen_territories(dataset, kwargs['rng_seed'])
        cfg = RegionConfig(**kwargs)
        _REGION_CACHE[key] = cfg
        return cfg

    def _autogen_territories(dataset: ProjectDataset, rng_seed: int) -> tuple[list, list, list]:
        """D-03: synthesize Condado_NNN from feature centroids when YAML omits territory data.
        Every condado gets a unique `original_idx` (CLAUDE.md rule 4 — Nájera bug guard).
        All assigned to single 'unnamed' kingdom (color gray (128,128,128)).
        Determinism via rng_seed (currently unused beyond color jitter — kept for forward compat).
        """
        # Read GeoJSON, extract feature representative_points, build list of dicts.
        # Returns (kingdoms, duchies, condados) — kingdoms/duchies single 'unnamed' entry.
        # Implementation: open pt_geojson + es_input (json.load), iterate features,
        # shapely.geometry.shape(feature.geometry).representative_point() for centroid.
        # Each condado dict mirrors territory_data.py CONDADOS shape with original_idx
        # 1..N. Implementer fills based on observed RegionConfig.condados shape.
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user→loader | `key` string supplied by HTTP layer (D-06 endpoint, D-07 modal) or test code; may contain attack payloads |
| YAML→loader | YAML file bytes parsed; may contain malicious tags (`!!python/object/apply`) or oversized payloads |
| YAML→filesystem | `dataset.*` strings in YAML resolve to filesystem paths; may attempt traversal escape |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-01-01 | Tampering / Information Disclosure | `load_region(key)` | mitigate | Regex `^[a-z0-9_]+$` validates key before any filesystem access; reject `..`/`/`/`\\`/null bytes. Covered by acceptance grep + unit test cases. |
| T-05-01-02 | Tampering (RCE) | `region_loader.py` YAML parse | mitigate | `yaml.safe_load` only (NEVER `yaml.load`); acceptance criterion greps both ways. PyYAML safe loader rejects `!!python/object/*` tags. |
| T-05-01-03 | Information Disclosure | `dataset.*` path resolution | mitigate | YAML-relative paths resolved under `{regions_dir}/{key}/`; post-resolve `.relative_to(region_root)` guard rejects escape. Unit test with `../../../etc/passwd` payload. |
