    ```python
    """SC-3: France 1066 toy pipeline produces 12 well-formed Unity contract files.

    Parity to Reconquista is NOT required (CONTEXT.md SC-3 boundary).
    File presence + dimensions + JSON schema validity IS required.
    """
    import json
    import pytest
    from pathlib import Path
    from dataclasses import replace
    from PIL import Image
    from fastapi.testclient import TestClient  # only needed if going through the HTTP boundary
    from medieval_forge.services.pipeline.region_loader import load_region, clear_region_cache
    from medieval_forge.services.pipeline import run_pipeline  # sync def confirmed by read_first
    from medieval_forge.services.pipeline.contracts import EXPORT_FILE_CONTRACT  # R-14

    CONTRACT_DIMS = {
        "lookup_barony.png": (1920, 1080),
        "lookup_condado.png": (1920, 1080),
        "lookup_barony_colors.json": None,
        "lookup_condado_colors.json": None,
        "terrain_lookup.png": (1920, 1080),
        "terrain_types.json": None,
        "territory_metadata.json": None,
        "visual_condado.png": (3840, 2160),
        "visual_barony.png": (3840, 2160),
        "mountains_mask.png": (3840, 2160),
        "rivers_overlay.png": (3840, 2160),
        "mountain_river_data.json": None,
    }
    assert set(CONTRACT_DIMS) == set(EXPORT_FILE_CONTRACT), "12-file contract drifted — update EXPORT_FILE_CONTRACT"

    @pytest.mark.e2e
    def test_france_1066_export_contract(tmp_path: Path):
        clear_region_cache()
        cfg = replace(load_region("france_1066"), output_dir=str(tmp_path))
        run_pipeline(cfg)  # SYNC path
        _assert_contract(tmp_path)

    def _assert_contract(out: Path) -> None:
        present = {p.name for p in out.iterdir() if p.is_file()}
        missing = set(EXPORT_FILE_CONTRACT) - present
        assert not missing, f"missing contract files: {missing}"
        for name, dims in CONTRACT_DIMS.items():
            if dims is None:
                continue
            img = Image.open(out / name)
            assert img.size == dims, f"{name}: expected {dims}, got {img.size}"
        for name in EXPORT_FILE_CONTRACT:
            if name.endswith(".json"):
                data = json.loads((out / name).read_text(encoding="utf-8"))
                assert data is not None, f"{name}: empty"
        # CLAUDE.md rule 4: every condado must carry a unique original_idx
        meta = json.loads((out / "territory_metadata.json").read_text(encoding="utf-8"))
        idxs = [c["original_idx"] for c in _walk_condados(meta)]
        assert idxs, "territory_metadata.json contains no condados"
        assert len(idxs) == len(set(idxs)), f"original_idx not unique: {idxs}"

    def _walk_condados(meta):
        # Adapt to observed Iberia shape: meta may be dict with 'condados' or nested.
        if isinstance(meta, dict) and "condados" in meta:
            return meta["condados"]
        if isinstance(meta, list):
            return [m for m in meta if isinstance(m, dict) and "original_idx" in m]
        raise AssertionError(f"unknown territory_metadata.json shape: {type(meta).__name__}")
    ```
