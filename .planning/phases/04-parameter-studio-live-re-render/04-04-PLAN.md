// CLAUDE.md non-negotiable: Konva.clearCache() after every geometric mutation.
// Phase 04: cacheVersion or stageView change = geometric mutation. Pitfall 6:

useEffect(() => {
  const stage = stageRef.current
  if (!stage) return
  // findAll('Layer') returns every Konva.Layer in the stage in z-order.

    const layer = node as Konva.Layer
    layer.clearCache()
    layer.batchDraw()
  })
  // Dependencies: effectiveCacheVersion + stageView. effectiveCacheVersion folds
  // priorTokens.render so cancel triggers a re-cache bust per D-13 + D-14.
