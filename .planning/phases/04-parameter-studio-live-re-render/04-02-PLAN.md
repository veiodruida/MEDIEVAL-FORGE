"""Shared run-state dicts for v3 SSE endpoints (generate + render).

Module-level singletons. Both api/v3/generate.py and api/v3/render.py import
from here so the cross-router 409 single-flight gate (Plan 04-02 Task 2) can
inspect both task slots atomically.

Layout:
  _RUN_QUEUES[project_id]      -> asyncio.Queue[str | None]   (SSE producer)
  _RUN_TASKS[project_id]       -> asyncio.Task                 (the producer task)
  _RUN_STOP_EVENTS[project_id] -> threading.Event              (D-14 cancel signal)
  _RUN_KIND[project_id]        -> 'generate' | 'render'        (for diagnostics + 409 detail)

Single-flight invariant: at most ONE alive task per project across BOTH dicts.
is_run_alive() is the canonical check.
"""
from __future__ import annotations

import asyncio
import threading
from typing import Optional


_RUN_QUEUES: dict[str, asyncio.Queue[Optional[str]]] = {}
_RUN_TASKS: dict[str, asyncio.Task] = {}
_RUN_STOP_EVENTS: dict[str, threading.Event] = {}
_RUN_KIND: dict[str, str] = {}


def is_run_alive(project_id: str) -> Optional[str]:
    """Return 'generate' | 'render' if a task for that project is still alive,
    else None. Cross-router single-flight gate (D-04)."""
    task = _RUN_TASKS.get(project_id)
    if task is not None and not task.done():
        return _RUN_KIND.get(project_id, "unknown")
    return None


__all__ = [
    "_RUN_QUEUES",
    "_RUN_TASKS",
    "_RUN_STOP_EVENTS",
    "_RUN_KIND",
    "is_run_alive",
]
```

def run_pipeline_incremental(cfg: RegionConfig, project_id: str) -> list[str]:
    """DAG-walking variant of run_pipeline.

    1. Compute new token for each stage in DAG_ORDER.
    2. For each stage where new_token != cached token: recompute, cache_put.
    3. Skip stages where token matches (their array is already cached).
    4. After all affected stages complete, re-emit the file outputs (lookup PNGs +
       visual PNGs + sidecars) so the canvas hydrators see new artifacts.

    Returns the list of stage names that were recomputed (for the SSE
    producer to surface as `affected_stages`). Raises StageCancelled if
    cfg.stop_event is set mid-run; caller catches and emits stage_cancel.
    """
    from .cache import cache_get, cache_put
    from .dag import DAG_ORDER, DAG_PARENTS, STAGE_READS, compute_version_token
    from .cleanup import (
        apply_median, remove_fragments, smooth_per_territory, merge_small_blobs,
    )
    from .landmask import load_municipalities, build_land_mask
    from .border import build_border_mask
    from .voronoi import setup_baronies, rasterize_baronies, build_hierarchy_maps
    from .render import render_map, render_mountains, render_rivers
    from .lookup import generate_lookup_map
    from .export import export_metadata
    from ..canvas_sidecars import (
        build_baronies_geojson_sidecar,
        build_territories_geojson_sidecar,
    )

    affected: list[str] = []
    tokens: dict[str, str] = {}

    # Helper: check if a stage's new_token differs from the cached token.
    def stage_dirty(stage: str) -> bool:
        upstream = [tokens[p] for p in DAG_PARENTS[stage]]
        new_tok = compute_version_token(stage, STAGE_READS[stage], cfg, upstream)
        tokens[stage] = new_tok
        cached = cache_get(project_id, stage)
        return cached is None or cached.token != new_tok

    # Walk DAG. We always need landmask + border + voronoi cached for downstream
    # to work, so on first run (cold cache) every stage runs. Subsequent /render
    # only recomputes stages whose reads changed.

    if stage_dirty("landmask"):
        affected.append("landmask")
        if cfg.on_stage: cfg.on_stage("landmask", "start")
        # NOTE: build_land_mask returns 1x land + we also need land_2x for render.
        # For Phase 04, cache the 1x land array; the 2x is recomputed in render
        # stage if render is affected (cheap relative to cleanup).
        pt_data, es_municipalities = load_municipalities(cfg)
        land = build_land_mask(pt_data, es_municipalities, cfg)
        cache_put(project_id, "landmask", tokens["landmask"], land)
        if cfg.on_stage: cfg.on_stage("landmask", "done")
    else:
        land = cache_get(project_id, "landmask").array
        # Re-load municipalities (not cached — cheap re-parse)
        pt_data, es_municipalities = load_municipalities(cfg)

    # ... similar for "border", "voronoi" (from raw rasterization) ...
    # See full skeleton in the cleanup-stage block below for the canonical pattern.

    # Cleanup: 4 stages, the meat of incremental re-render.
    if stage_dirty("median"):
        affected.append("median")
        if cfg.on_stage: cfg.on_stage("median", "start")
        med = apply_median(raw, land, nb, cfg)
        cache_put(project_id, "median", tokens["median"], med)
        if cfg.on_stage: cfg.on_stage("median", "done")
    else:
        med = cache_get(project_id, "median").array

    if stage_dirty("fragment"):
        affected.append("fragment")
        if cfg.on_stage: cfg.on_stage("fragment", "start")
        frag = remove_fragments(med, land, nb, cfg)
        cache_put(project_id, "fragment", tokens["fragment"], frag)
        if cfg.on_stage: cfg.on_stage("fragment", "done")
    else:
        frag = cache_get(project_id, "fragment").array

    if stage_dirty("smooth"):
        affected.append("smooth")
        if cfg.on_stage: cfg.on_stage("smooth", "start")
        sm = smooth_per_territory(frag, land, cfg)
        cache_put(project_id, "smooth", tokens["smooth"], sm)
        if cfg.on_stage: cfg.on_stage("smooth", "done")
    else:
        sm = cache_get(project_id, "smooth").array

    if stage_dirty("merge"):
        affected.append("merge")
        if cfg.on_stage: cfg.on_stage("merge", "start")
        result = merge_small_blobs(sm, land, nb, cfg)
        cache_put(project_id, "merge", tokens["merge"], result)
        if cfg.on_stage: cfg.on_stage("merge", "done")
    else:
        result = cache_get(project_id, "merge").array

    # ... continue for hierarchy, render, lookup, metadata, export ...

    # If any cleanup-or-later stage ran, write the 12-file Unity output.
    # (For Phase 04 default mode: always write outputs after a /render call so
    # canvas hydrators see fresh artifacts. Disk-write cost is dominated by the
    # cleanup compute we just did.)
    if affected:
        # Re-emit visual PNGs, lookup PNGs, sidecars to cfg.output_dir.
        # The full file-write block from run_pipeline (lines 145-235) is reused
        # verbatim. Extract a helper `_write_outputs_to_disk(cfg, result, ...)`
        # in `__init__.py` and call it from BOTH run_pipeline and
        # run_pipeline_incremental to guarantee parity.

    return affected


__all__ = ["run_pipeline", "run_pipeline_incremental"]
```

    smooth_sigma: Optional[float] = Field(default=None, ge=3.0, le=4.5)
    median_passes: Optional[int] = Field(default=None, ge=1, le=12)
    fragment_min_px: Optional[int] = Field(default=None, ge=0, le=2000)
    blob_merge_px: Optional[int] = Field(default=None, ge=0, le=500)


class RenderRequest(BaseModel):
    cfg_overrides: CfgOverrides = Field(default_factory=CfgOverrides)
    # stage_view is client-only (Pitfall 8). Kept here only for symmetry with the
    # frontend body; the server ignores it for token derivation.
    stage_view: Optional[str] = None

    model_config = {"extra": "forbid"}  # ASVS V5 reject unknown fields


def _emit(queue, event_type, stage, message="", progress=None, token=None):
    payload = {
        "event_type": event_type,
        "stage": stage,
        "message": message,
        "progress": progress,
        "token": token,
    }
    queue.put_nowait(f"data: {json.dumps(payload)}\n\n")


def _make_on_stage(queue, loop):
    """Same shape as generate._make_on_stage but with token field plumbed."""
    def on_stage(stage: str, evt: str) -> None:
        event_type = f"stage_{evt}"
        # Read latest cached token if available (set by run_pipeline_incremental
        # right before the 'done' emit; for 'start' it's the new token from the DAG walk).
        loop.call_soon_threadsafe(_emit, queue, event_type, stage, "OK", None, None)
    return on_stage


async def _render_producer(
    project_id: str,
    overrides: dict,
    queue,
    sf,
) -> None:
    """Producer task. Always puts None sentinel before returning. On
    StageCancelled, emits stage_cancel events with prior_token from cache."""
    stop_event = threading.Event()
    _RUN_STOP_EVENTS[project_id] = stop_event
    affected: list[str] = []
    try:
        _emit(queue, "started", None, f"Render iniciado para projeto {project_id}", 0.0)

        # Build fresh cfg per D-18.
        cfg = iberia_config()
        cfg.output_dir = str(project_dir(project_id) / "output")
        cfg.stop_event = stop_event
        # Apply overrides
        for k, v in overrides.items():
            if v is not None:
                setattr(cfg, k, v)
        cfg.on_stage = _make_on_stage(queue, asyncio.get_running_loop())

        affected = await asyncio.to_thread(
            run_pipeline_incremental, cfg, project_id,
        )
        _emit(queue, "done", None, f"OK affected={','.join(affected)}", 1.0)
    except StageCancelled as exc:
        # D-13: emit stage_cancel per affected stage, carrying prior_token in message.
        for stage in affected + [exc.stage_name]:
            entry = cache_get(project_id, stage)
            prior_tok = entry.prior_token if (entry and entry.prior_token) else ""
            _emit(queue, "stage_cancel", stage, prior_tok, None, prior_tok)
        _emit(queue, "done", None, "cancelled", 1.0)
    except Exception as exc:  # noqa: BLE001
        logger.exception("v3 render failed for project %s", project_id)
        _emit(queue, "error", None, exc.__class__.__name__, None)
    finally:
        await queue.put(None)
        _RUN_STOP_EVENTS.pop(project_id, None)


@router.post("/{project_id}/render", status_code=202)
async def trigger_render(
    project_id: str,
    body: RenderRequest,
    db: AsyncSession = Depends(get_db),
) -> dict:
    if not is_valid_uuid(project_id):
        raise HTTPException(status_code=400, detail="project_id must be a valid UUID")
    project = await db.get(Project, project_id)
    if project is None:
        raise HTTPException(status_code=404, detail="project not found")

    # Cross-router single-flight (D-04).
    alive_kind = is_run_alive(project_id)
    if alive_kind is not None:
        raise HTTPException(
            status_code=409,
            detail=f"project is already {alive_kind}; subscribe to /{alive_kind}/stream",
        )

    # Schedule producer
    queue: asyncio.Queue = asyncio.Queue()
    _RUN_QUEUES[project_id] = queue
    _RUN_KIND[project_id] = "render"
    overrides = body.cfg_overrides.model_dump(exclude_none=True)
    task = asyncio.create_task(
        _render_producer(project_id, overrides, queue, AsyncSessionLocal),
    )
    _RUN_TASKS[project_id] = task

    return {"run_id": str(uuid.uuid4()), "status": "scheduled", "kind": "render"}


@router.get("/{project_id}/render/stream")
async def stream_render(project_id: str) -> StreamingResponse:
    if not is_valid_uuid(project_id):
        raise HTTPException(status_code=400, detail="project_id must be a valid UUID")
    queue = _RUN_QUEUES.get(project_id)
    if queue is None:
        raise HTTPException(status_code=404, detail="no active render run; POST /render first")

    async def gen() -> AsyncIterator[str]:
        while True:
            msg = await queue.get()
            if msg is None:
                break
            yield msg
    return StreamingResponse(
        gen(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no", "Connection": "keep-alive"},
    )


@router.post("/{project_id}/render/cancel", status_code=200)
async def cancel_render(project_id: str) -> dict:
    if not is_valid_uuid(project_id):
        raise HTTPException(status_code=400, detail="project_id must be a valid UUID")
    stop_event = _RUN_STOP_EVENTS.get(project_id)
    if stop_event is None:
        raise HTTPException(status_code=404, detail="no active render run to cancel")
    stop_event.set()
    return {"status": "cancel_requested"}

def _array_to_png(arr: np.ndarray, stage_name: str) -> bytes:
    """Hand-rolled tab20 colormap (no matplotlib dep). bool → grayscale; int16 → cycle 20 colors."""
    if arr.dtype == np.bool_ or stage_name == "landmask":
        img = Image.fromarray((arr * 255).astype(np.uint8), mode="L")
    else:
        # tab20 palette (matplotlib values, hardcoded — Pitfall A3 fallback)
        TAB20 = np.array([
            [31, 119, 180], [174, 199, 232], [255, 127, 14], [255, 187, 120],
            [44, 160, 44], [152, 223, 138], [214, 39, 40], [255, 152, 150],
            [148, 103, 189], [197, 176, 213], [140, 86, 75], [196, 156, 148],
            [227, 119, 194], [247, 182, 210], [127, 127, 127], [199, 199, 199],
            [188, 189, 34], [219, 219, 141], [23, 190, 207], [158, 218, 229],
        ], dtype=np.uint8)
        h, w = arr.shape
        rgb = np.zeros((h, w, 3), dtype=np.uint8)
        valid = arr >= 0
        idx = (arr[valid] % 20).astype(int)
        rgb[valid] = TAB20[idx]
        img = Image.fromarray(rgb, mode="RGB")
    buf = io.BytesIO()
    img.save(buf, format="PNG")
    return buf.getvalue()
