    Step A — Create `backend/medieval_forge/services/llm/auth.py`:
    ```python
    from __future__ import annotations
    import json
    import os
    import pathlib
    import time
    from typing import Any, Literal

    AUTH_FILE_CANDIDATES = [
        pathlib.Path.home() / ".claude" / ".credentials.json",
        pathlib.Path(os.environ.get("APPDATA", "")) / "Claude" / "credentials.json" if os.environ.get("APPDATA") else None,
        pathlib.Path(os.environ.get("XDG_CONFIG_HOME", str(pathlib.Path.home() / ".config"))) / "claude" / "credentials.json",
    ]


    def read_claude_cli_token() -> str | None:
        """Read Anthropic OAuth access token from claude-code credential store.

        Returns None if file missing, malformed, or token expired.
        Verified file structure (RESEARCH.md Pattern 4):
            {"claudeAiOauth": {"accessToken": "sk-ant-oat01-...", "expiresAt": 1776790173634, ...}}
        """
        for path in AUTH_FILE_CANDIDATES:
            if path is None or not path.exists():
                continue
            try:
                data = json.loads(path.read_text(encoding="utf-8"))
            except (json.JSONDecodeError, OSError):
                continue
            token_data = data.get("claudeAiOauth", {})
            access_token = token_data.get("accessToken")
            expires_at_ms = token_data.get("expiresAt", 0)
            if access_token and expires_at_ms > time.time() * 1000:
                return access_token
        return None


    def resolve_credentials(provider_id: str, app_state: Any) -> dict[str, Any]:
        """Priority chains per provider (D-10..D-13)."""
        session = (getattr(app_state, "credentials", {}) or {}).get(provider_id)

        if provider_id == "claude":
            if session:
                return {**session, "source": session.get("source", "session")}
            env_key = os.getenv("ANTHROPIC_API_KEY")
            if env_key:
                return {"type": "api_key", "key": env_key, "source": "env"}
            cli_token = read_claude_cli_token()
            if cli_token:
                return {"type": "oauth_token", "key": cli_token, "source": "cli"}
            return {}

        if provider_id == "gemini":
            if session:
                return {**session, "source": session.get("source", "session")}
            env_key = os.getenv("GOOGLE_API_KEY") or os.getenv("GEMINI_API_KEY")
            if env_key:
                return {"type": "api_key", "api_key": env_key, "source": "env"}
            return {}

        if provider_id == "openai":
            if session:
                return {**session, "source": session.get("source", "session")}
            env_key = os.getenv("OPENAI_API_KEY")
            if env_key:
                return {"type": "api_key", "key": env_key, "source": "env"}
            return {}

        if provider_id == "ollama":
            return {"type": "none", "source": "none"}

        return {}
    ```

    ```python
    from __future__ import annotations
    import secrets
    import time
    from typing import Literal

    from fastapi import APIRouter, HTTPException, Request, Response
    from pydantic import BaseModel

    from ..services.llm.registry import PROVIDERS

    router = APIRouter(tags=["auth"])

    OAUTH_STATE_TTL_SEC = 300

    # OAuth client config — bundled for the local desktop tool (Open Question #2).
    # NOTE: replace placeholders with real GCP-registered Desktop OAuth client before shipping.
    GOOGLE_CLIENT_CONFIG = {
        "installed": {
            "client_id": "REPLACE_WITH_BUNDLED_CLIENT_ID.apps.googleusercontent.com",
            "client_secret": "REPLACE_WITH_BUNDLED_CLIENT_SECRET",
            "redirect_uris": ["http://localhost:8000/api/auth/oauth/gemini/callback"],
            "auth_uri": "https://accounts.google.com/o/oauth2/auth",
            "token_uri": "https://oauth2.googleapis.com/token",
        }
    }
    GEMINI_SCOPES = ["https://www.googleapis.com/auth/generative-language.retriever"]
    OAUTH_REDIRECT_URI = "http://localhost:8000/api/auth/oauth/gemini/callback"


    class CredentialPayload(BaseModel):
        api_key: str


    def _clean_expired_states(app_state) -> None:
        """Pitfall 5: drop OAuth states older than TTL."""
        now = time.time()
        states = getattr(app_state, "oauth_states", {})
        expired = [k for k, v in states.items() if v.get("expires_at", 0) < now]
        for k in expired:
            states.pop(k, None)


    @router.post("/auth/credentials/{provider}", status_code=204)
    async def store_credential(provider: str, payload: CredentialPayload, request: Request) -> Response:
        if provider not in PROVIDERS:
            raise HTTPException(404, f"Unknown provider: {provider}")
        # Gemini stores api_key under "api_key" field; others under "key" (matches what providers expect)
        creds = getattr(request.app.state, "credentials", None)
        if creds is None:
            creds = {}
            request.app.state.credentials = creds
        if provider == "gemini":
            creds[provider] = {"type": "api_key", "api_key": payload.api_key, "source": "session"}
        else:
            creds[provider] = {"type": "api_key", "key": payload.api_key, "source": "session"}
        return Response(status_code=204)


    @router.delete("/auth/credentials/{provider}", status_code=204)
    async def clear_credential(provider: str, request: Request) -> Response:
        if provider not in PROVIDERS:
            raise HTTPException(404, f"Unknown provider: {provider}")
        creds = getattr(request.app.state, "credentials", {}) or {}
        creds.pop(provider, None)
        return Response(status_code=204)


    @router.post("/auth/oauth/{provider}/start")
    async def oauth_start(provider: str, request: Request) -> dict:
        if provider != "gemini":
            raise HTTPException(404, f"OAuth not supported for provider: {provider}")
        from google_auth_oauthlib.flow import Flow

        states = getattr(request.app.state, "oauth_states", None)
        if states is None:
            states = {}
            request.app.state.oauth_states = states
        _clean_expired_states(request.app.state)

        flow = Flow.from_client_config(
            GOOGLE_CLIENT_CONFIG,
            scopes=GEMINI_SCOPES,
            redirect_uri=OAUTH_REDIRECT_URI,
        )
        state = secrets.token_urlsafe(32)
        authorize_url, _ = flow.authorization_url(state=state, access_type="offline", prompt="consent")
        states[state] = {
            "flow": flow,
            "provider": provider,
            "expires_at": time.time() + OAUTH_STATE_TTL_SEC,
        }
        return {"authorize_url": authorize_url, "state": state}


    @router.get("/auth/oauth/{provider}/callback")
    async def oauth_callback(provider: str, code: str, state: str, request: Request) -> dict:
        if provider != "gemini":
            raise HTTPException(404, f"OAuth not supported for provider: {provider}")
        states = getattr(request.app.state, "oauth_states", {}) or {}
        stored = states.pop(state, None)
        if stored is None:
            raise HTTPException(400, "Invalid or expired OAuth state")
        if stored.get("expires_at", 0) < time.time():
            raise HTTPException(400, "Expired OAuth state")
        flow = stored["flow"]
        flow.fetch_token(code=code)
        creds = getattr(request.app.state, "credentials", None)
        if creds is None:
            creds = {}
            request.app.state.credentials = creds
        creds[provider] = {
            "type": "oauth",
            "access_token": flow.credentials.token,
            "refresh_token": flow.credentials.refresh_token,
            "source": "oauth",
        }
        return {"status": "authenticated", "provider": provider}
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → /api/auth/credentials/* | User-supplied API key crosses HTTP boundary into in-memory store |
| Browser → /api/auth/oauth/{provider}/callback | OAuth code arrives from external Google redirect |
| Local filesystem → backend (CLI piggyback) | Backend reads `~/.claude/.credentials.json` written by external `claude-code` tool |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-3-01 | Information Disclosure | API key persistence | mitigate | Keys stored only in `app.state.credentials` (in-process dict); zero `open(..., "w")` calls in auth modules — verified by acceptance_criteria grep; D-14 |
| T-3-02 | Spoofing (CSRF) | OAuth callback | mitigate | `secrets.token_urlsafe(32)` state param required; unknown/expired state → 400; state TTL 300s with cleanup on each /start (Pitfall 5) |
| T-3-03 | Information Disclosure | API key echoed in response body | mitigate | POST returns 204 No Content; test asserts response body does not contain submitted key |
| T-3-04 | Tampering | OAuth `client_secret` exposed | mitigate | client_secret kept in backend Python module only (not frontend bundle); fetched secret used server-side via google-auth-oauthlib only |
| T-3-05 | Information Disclosure | CLI credential file contents leaked via logs | mitigate | `read_claude_cli_token` returns only the token string; never logs file contents; no `print()` or `logger.*` calls inside the function |
| T-3-06 | Denial of Service | OAuth state dict memory leak | mitigate | TTL cleanup on every /start call (Pitfall 5); abandoned flows expire in 5 min |
| T-3-07 | Spoofing | Forged Authorization header to /credentials | accept | Local-only single-user tool; no auth on /api/* by design; documented in REQUIREMENTS Out of Scope ("single-user local tool") |

</threat_model>
