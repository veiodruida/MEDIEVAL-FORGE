    ```tsx
    import { Layer, Image as KonvaImage } from 'react-konva'
    import useImage from 'use-image'
    import { useUIStore } from '../../stores/uiStore'

    interface Props {
      mapW: number
      mapH: number
    }

    export function TerrainOverlayLayer({ mapW, mapH }: Props): JSX.Element | null {
      const overlayImageUrl = useUIStore((s) => s.overlayImageUrl)
      const overlayOpacity = useUIStore((s) => s.overlayOpacity)
      const [img, status] = useImage(overlayImageUrl ?? '')

      if (!overlayImageUrl || status !== 'loaded' || !img) {
        return null
      }

      return (
        <Layer listening={false}>
          <KonvaImage
            image={img}
            x={0}
            y={0}
            width={mapW}
            height={mapH}
            opacity={overlayOpacity}
            listening={false}
          />
        </Layer>
      )
    }
    ```

    ```tsx
    import { useRef, useState } from 'react'
    import { Card, Flex, Text, Button, Slider } from '@radix-ui/themes'
    import { useUIStore } from '../../stores/uiStore'
    import { useEditorStore } from '../../stores/useEditorStore'
    // Reuse existing toast helper. Inspect a sibling component (e.g. CanvasViewer)
    // for the exact import path for the toast utility used in this project.
    import { toast } from '<existing toast module — replace with actual path found via read_first>'

    export function ReferenceOverlayPanel(): JSX.Element | null {
      const editMode = useEditorStore((s) => s.editMode)
      const overlayImageUrl = useUIStore((s) => s.overlayImageUrl)
      const overlayOpacity = useUIStore((s) => s.overlayOpacity)
      const setOverlayImageUrl = useUIStore((s) => s.setOverlayImageUrl)
      const setOverlayOpacity = useUIStore((s) => s.setOverlayOpacity)

      const fileInputRef = useRef<HTMLInputElement>(null)
      const [filename, setFilename] = useState<string | null>(null)

      if (!editMode) return null

      const onFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
        const file = e.target.files?.[0]
        if (!file) return
        if (!file.type.startsWith('image/')) {
          toast.error('Arquivo inválido. Carregue uma imagem (PNG, JPEG, WebP).', { duration: 6000 })
          e.target.value = ''
          return
        }
        const url = URL.createObjectURL(file)
        setOverlayImageUrl(url)
        setFilename(file.name)
        e.target.value = ''
      }

      const onClear = () => {
        setOverlayImageUrl(null)
        setFilename(null)
      }

      return (
        <Card
          variant="surface"
          style={{ position: 'absolute', bottom: 52, left: 12, width: 200, zIndex: 10 }}
        >
          <Flex direction="column" gap="2">
            <Text size="2" weight="bold">Sobreposição</Text>
            <input
              ref={fileInputRef}
              type="file"
              accept="image/*"
              style={{ display: 'none' }}
              onChange={onFileChange}
            />
            <Button size="1" variant="soft" onClick={() => fileInputRef.current?.click()}>
              Carregar imagem…
            </Button>
            {filename && <Text size="1" color="gray">{filename}</Text>}
            <Flex direction="column" gap="1">
              <Flex justify="between" align="center">
                <Text size="1">Opacidade</Text>
                <Text size="1" color="gray">{Math.round(overlayOpacity * 100)}%</Text>
              </Flex>
              <Slider
                size="1"
                min={0}
                max={1}
                step={0.05}
                value={[overlayOpacity]}
                onValueChange={(v) => setOverlayOpacity(v[0])}
              />
            </Flex>
            {overlayImageUrl !== null && (
              <Button size="1" variant="ghost" color="gray" onClick={onClear}>
                Remover
              </Button>
            )}
          </Flex>
        </Card>
      )
    }
    ```
