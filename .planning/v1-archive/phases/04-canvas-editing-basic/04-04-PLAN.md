    ```python
    # ==================== Phase 4: Edit API schemas ====================
    from pydantic import BaseModel, Field, field_validator
    from typing import Literal


    class MoveCapitalRequest(BaseModel):
        lon: float = Field(..., ge=-180.0, le=180.0, description="Longitude in degrees")
        lat: float = Field(..., ge=-90.0, le=90.0, description="Latitude in degrees")


    class MoveCapitalResponse(BaseModel):
        updated_territories: dict[str, dict]   # id -> GeoJSON Polygon/MultiPolygon
        affected_ids: list[str]


    class MergeRequest(BaseModel):
        condado_ids: list[str] = Field(..., min_length=2, description="IDs to merge; primary_id MUST be in this list")
        primary_id: str = Field(..., description="ID whose name and key the merged result inherits")

        @field_validator("condado_ids")
        @classmethod
        def _unique_ids(cls, v: list[str]) -> list[str]:
            if len(set(v)) != len(v):
                raise ValueError("condado_ids must be unique")
            return v


    class MergeResponse(BaseModel):
        merged_id: str
        merged_territory: dict                   # GeoJSON (Polygon or MultiPolygon)
        removed_ids: list[str]
        warning: Literal["non_adjacent_multipolygon"] | None = None


    class SplitRequest(BaseModel):
        cut_line: list[tuple[float, float]] = Field(..., min_length=2, max_length=1000)
        mode: Literal["snap", "polyline", "freehand"]

        @field_validator("cut_line")
        @classmethod
        def _coord_bounds(cls, v: list[tuple[float, float]]) -> list[tuple[float, float]]:
            for lon, lat in v:
                if not (-180.0 <= lon <= 180.0) or not (-90.0 <= lat <= 90.0):
                    raise ValueError(f"cut_line coordinate out of range: ({lon}, {lat})")
            return v


    class SplitResponse(BaseModel):
        original_id: str
        new_territory_a: dict                    # {id, geometry}
        new_territory_b: dict


    class ReshapeGeometryRequest(BaseModel):
        geometry: dict                           # GeoJSON Polygon

        @field_validator("geometry")
        @classmethod
        def _polygon_size_bound(cls, v: dict) -> dict:
            # DoS bound per Research §Security Domain
            if v.get("type") != "Polygon":
                raise ValueError("geometry.type must be 'Polygon'")
            coords = v.get("coordinates") or []
            if not coords or not coords[0]:
                raise ValueError("geometry.coordinates[0] (exterior ring) required")
            if len(coords[0]) > 10_000:
                raise ValueError("polygon exterior has >10,000 coordinates — rejected (DoS bound)")
            return v
    ```

    ```python
    @router.post(
        "/projects/{project_id}/territories/{condado_id}/recalc",
        response_model=MoveCapitalResponse,
    )
    async def move_capital(
        project_id: str,
        condado_id: str,
        body: MoveCapitalRequest,
        session: AsyncSession = Depends(get_session),
    ) -> MoveCapitalResponse:
        from ..services import voronoi as voro_svc
        from ..services.territories_geojson import load_territories, save_territories

        territories = await load_territories(project_id)  # returns {id: {geometry, lon, lat, ...}}
        if condado_id not in territories:
            raise HTTPException(status_code=404, detail=f"condado {condado_id} not found")

        # Build ordered seed list + id index map
        ids = list(territories.keys())
        seeds = [(territories[i].get("lon"), territories[i].get("lat")) for i in ids]

        # Replace moved seed with new coordinates
        moved_idx = ids.index(condado_id)
        seeds[moved_idx] = (body.lon, body.lat)

        try:
            updated = voro_svc.recalc_neighbors(seeds, moved_idx=moved_idx, land_mask_polygon=None)
        except ValueError as e:
            raise HTTPException(status_code=422, detail=str(e))

        # Map result back to {id: GeoJSON}
        from shapely.geometry import mapping
        updated_territories = {ids[i]: mapping(geom) for i, geom in updated.items()}
        affected_ids = list(updated_territories.keys())

        # Persist: update capital + territory geoms
        territories[condado_id]["lon"] = body.lon
        territories[condado_id]["lat"] = body.lat
        for aid in affected_ids:
            territories[aid]["geometry"] = updated_territories[aid]
        await save_territories(project_id, territories)

        return MoveCapitalResponse(
            updated_territories=updated_territories,
            affected_ids=affected_ids,
        )
    ```

    **B — `merge_territories_endpoint`** (EDIT-03):

    ```python
    @router.post(
        "/projects/{project_id}/territories/merge",
        response_model=MergeResponse,
    )
    async def merge_territories_endpoint(
        project_id: str,
        body: MergeRequest,
        session: AsyncSession = Depends(get_session),
    ) -> MergeResponse:
        from ..services import voronoi as voro_svc
        from ..services.territories_geojson import load_territories, save_territories
        from shapely.geometry import shape, mapping

        if body.primary_id not in body.condado_ids:
            raise HTTPException(status_code=422, detail="primary_id must be in condado_ids")

        territories = await load_territories(project_id)
        missing = [cid for cid in body.condado_ids if cid not in territories]
        if missing:
            raise HTTPException(status_code=404, detail=f"condado(s) not found: {missing}")

        polys = [shape(territories[cid]["geometry"]) for cid in body.condado_ids]
        try:
            merged, warning = voro_svc.merge_territories(polys)
        except ValueError as e:
            raise HTTPException(status_code=422, detail=str(e))

        merged_geojson = mapping(merged)
        removed_ids = [cid for cid in body.condado_ids if cid != body.primary_id]

        # Persist: replace primary with merged geom; drop removed ids
        territories[body.primary_id]["geometry"] = merged_geojson
        for rid in removed_ids:
            territories.pop(rid, None)
        await save_territories(project_id, territories)

        # Adjacency is rebuilt on-the-fly in the next recalc call (Research §Pitfall 3 + Open Q 2)

        return MergeResponse(
            merged_id=body.primary_id,
            merged_territory=merged_geojson,
            removed_ids=removed_ids,
            warning=warning,
        )
    ```

    @router.post(

        project_id: str,
        condado_id: str,
        body: SplitRequest,
        session: AsyncSession = Depends(get_session),
    ) -> SplitResponse:
        import uuid
        from ..services import voronoi as voro_svc
        from ..services.territories_geojson import load_territories, save_territories
        from shapely.geometry import shape, mapping, LineString

        territories = await load_territories(project_id)
        if condado_id not in territories:

        poly = shape(territories[condado_id]["geometry"])
        cut_line = LineString(body.cut_line)

        try:
            halves = voro_svc.split_territory(poly, cut_line)  # raises ValueError if non-bisecting
        except ValueError as e:
            # Pitfall 4: must be 422, not 500. Frontend shows toast from detail.
            raise HTTPException(status_code=422, detail=str(e))

        if len(halves) < 2:
            raise HTTPException(status_code=422, detail="split produced fewer than 2 polygons")

        # Original inherits half[0]; generate new id for half[1]
        new_a_id = condado_id
        new_b_id = f"{condado_id}-split-{uuid.uuid4().hex[:8]}"

        territories[new_a_id]["geometry"] = mapping(halves[0])
        territories[new_b_id] = {
            **territories[condado_id],
            "geometry": mapping(halves[1]),
            "id": new_b_id,
            "name": territories[condado_id].get("name", "") + " (split)",
        }
        await save_territories(project_id, territories)

        return SplitResponse(
            original_id=condado_id,
            new_territory_a={"id": new_a_id, "geometry": mapping(halves[0])},
            new_territory_b={"id": new_b_id, "geometry": mapping(halves[1])},

    ) -> VertexHandlesResponse:

        poly = shape(territories[condado_id]["geometry"])
        if poly.geom_type != "Polygon":

        decimated = voro_svc.decimate_polygon(poly, target_vertices=target)
        decimated_coords = list(decimated.exterior.coords)[:-1]  # drop closing duplicate

        # Map each decimated vertex to its nearest index in the ORIGINAL exterior ring
        original_coords = list(poly.exterior.coords)[:-1]
        handles: list[VertexHandle] = []
        for dx, dy in decimated_coords:
            best_idx = 0
            best_dist_sq = float("inf")
            for i, (ox, oy) in enumerate(original_coords):
                d_sq = (ox - dx) ** 2 + (oy - dy) ** 2
                if d_sq < best_dist_sq:
                    best_dist_sq = d_sq
                    best_idx = i
            handles.append(VertexHandle(lon=dx, lat=dy, source_index=best_idx))

        return VertexHandlesResponse(handles=handles)
